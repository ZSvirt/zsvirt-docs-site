title: "新建虚拟机"

<APIPage method="POST" path="/v1/vm-instances">

创建一个新的虚拟机。

Headers

```
Authorization: OAuth the-session-uuid
```

Body

```

    "name": "vm1",
    "l3NetworkUuids": [

    "clusterUuid": "a0468dc645223f67bd0f2ab95276bbae",
    "description": "this is a vm",
    "strategy": "InstantStart",
    "diskAOs": [
      {
        "boot": true,
        "platform": "Linux",
        "guestOsType": "Helix 8",
        "architecture": "x86_64",
        "primaryStorageUuid": "ff0359abb8443d2f8077cc368afa7c00",
        "size": 0,
        "sourceType": "VolumeVO",
        "sourceUuid": "ff0eb5a723e9316bb1053ed7e4bb3100",
        "systemTags": [
          "volumeProvisioningStrategy::ThickProvisioning"
        ],
        "name": "root-volume"
      },
```

</Callout>
